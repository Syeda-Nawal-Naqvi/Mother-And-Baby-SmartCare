rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() {
      return request.auth != null;
    }

    function isSelf(uid) {
      return isSignedIn() && request.auth.uid == uid;
    }

    function callerDoc() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid));
    }

    function callerDocExists() {
      return exists(/databases/$(database)/documents/users/$(request.auth.uid));
    }

    function isAdmin() {
      return isSignedIn() && callerDocExists() &&
        callerDoc().data.role == 'admin' &&
        callerDoc().data.blocked != true;
    }

    function isNotBlocked() {
      return !callerDocExists() || callerDoc().data.blocked != true;
    }

    function isKnownUserSubcollection(name) {
      return name in [
        'babies', 'mother_profile', 'baby_medical_history', 'baby_weight',
        'blood_pressure', 'glucose', 'medical_history', 'milestones',
        'mother_weight', 'allergies', 'vaccinations', 'reminders'
      ];
    }

    function isSelfAssignableRole(role) {
      return role in ['mother', 'father', 'caretaker'];
    }

    match /users/{uid} {
      allow read: if isSelf(uid) || isAdmin();
      allow create: if isSelf(uid);
      allow update: if (isSelf(uid) &&
                          (
                            !('blocked' in request.resource.data.diff(resource.data).affectedKeys()) ||

                            (request.resource.data.blocked == true &&
                             resource.data.blocked != true &&
                             request.resource.data.blockedBy == 'self')
                          ) &&
                          (
                            !('role' in request.resource.data.diff(resource.data).affectedKeys()) ||
                            (resource.data.role != 'admin' &&
                             isSelfAssignableRole(request.resource.data.role))
                          ))
                        || isAdmin();
      allow delete: if isSelf(uid) || isAdmin();

      match /sessions/{sessionId} {
        allow read, write: if isSelf(uid) || isAdmin();
      }

      match /{subcollection}/{docId} {
        allow read, write: if isKnownUserSubcollection(subcollection) &&
                               isSelf(uid) && isNotBlocked();
        allow read, delete: if isKnownUserSubcollection(subcollection) && isAdmin();
      }
    }

    match /admin_uids/{uid} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }

    match /{path=**}/babies/{docId} {
      allow read, delete: if isAdmin();
    }
    match /{path=**}/mother_profile/{docId} {
      allow read, delete: if isAdmin();
    }
    match /{path=**}/baby_medical_history/{docId} {
      allow read, delete: if isAdmin();
    }
    match /{path=**}/baby_weight/{docId} {
      allow read, delete: if isAdmin();
    }
    match /{path=**}/blood_pressure/{docId} {
      allow read, delete: if isAdmin();
    }
    match /{path=**}/glucose/{docId} {
      allow read, delete: if isAdmin();
    }
    match /{path=**}/medical_history/{docId} {
      allow read, delete: if isAdmin();
    }
    match /{path=**}/milestones/{docId} {
      allow read, delete: if isAdmin();
    }
    match /{path=**}/mother_weight/{docId} {
      allow read, delete: if isAdmin();
    }
    match /{path=**}/allergies/{docId} {
      allow read, delete: if isAdmin();
    }
    match /{path=**}/vaccinations/{docId} {
      allow read, delete: if isAdmin();
    }
    match /{path=**}/reminders/{docId} {
      allow read, delete: if isAdmin();
    }

    match /feedback/{id} {
      allow create: if isSignedIn() &&
                        request.resource.data.userId == request.auth.uid;
      allow read: if isSignedIn() &&
                      (resource.data.userId == request.auth.uid || isAdmin());
      allow update: if isAdmin();
      allow delete: if isAdmin() ||
                        (isSignedIn() && resource.data.userId == request.auth.uid);
    }

    match /help_requests/{id} {
      allow create: if request.resource.data.keys().hasAll(
                        ['name', 'email', 'message', 'status', 'createdAt']) &&
                      request.resource.data.name is string &&
                      request.resource.data.name.size() > 0 &&
                      request.resource.data.name.size() < 200 &&
                      request.resource.data.email is string &&
                      request.resource.data.email.size() > 0 &&
                      request.resource.data.email.size() < 200 &&
                      request.resource.data.message is string &&
                      request.resource.data.message.size() > 0 &&
                      request.resource.data.message.size() < 3000 &&
                      request.resource.data.status == 'pending';
      allow read, update, delete: if isAdmin();
    }

    match /notifications/{id} {
      allow create: if isSignedIn();
      allow read, update: if isSignedIn() &&
                              (resource.data.recipientId == request.auth.uid || isAdmin());
      allow delete: if isSignedIn() &&
                        (resource.data.recipientId == request.auth.uid || isAdmin());
    }

    match /adminNotifications/{id} {
      allow create: if isSignedIn();
      allow read, update, delete: if isAdmin();
    }

    match /_pending_notifications/{id} {
      allow read, create, delete: if isSignedIn();
    }

    match /shop_products/{id} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }
    match /shop_categories/{id} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
